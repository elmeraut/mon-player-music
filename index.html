    <script>

        // Initialisation Telegram

        if (window.Telegram && window.Telegram.WebApp) {

            window.Telegram.WebApp.ready();

        }


        // Variables globales

        let mesPlaylists = {}; 

        const audio = document.getElementById('audioPlayer');

        const playlistsScreen = document.getElementById('playlists-screen');

        const tracksScreen = document.getElementById('tracks-screen');

        const playlistsList = document.getElementById('playlists-list');

        const tracksList = document.getElementById('tracks-list');

        const currentPlaylistTitle = document.getElementById('current-playlist-title');


        // 1. Chargement du fichier JSON généré par Python

        fetch('playlist.json')

            .then(response => {

                if (!response.ok) throw new Error("Fichier introuvable");

                return response.json();

            })

            .then(data => {

                mesPlaylists = data;

                showPlaylists(); // On affiche le menu une fois les données chargées

            })

            .catch(error => {

                console.error("Erreur:", error);

                playlistsList.innerHTML = "<p>Erreur : impossible de charger le fichier playlist.json. Vérifie qu'il est bien sur GitHub.</p>";

            });


        // 2. Fonction pour afficher la liste des playlists

        function showPlaylists() {

            playlistsScreen.style.display = 'block';

            tracksScreen.style.display = 'none';

            

            playlistsList.innerHTML = ''; 

            

            for (const nomPlaylist in mesPlaylists) {

                const card = document.createElement('div');

                card.className = 'playlist-card';

                card.innerText = nomPlaylist;

                card.onclick = () => openPlaylist(nomPlaylist);

                playlistsList.appendChild(card);

            }

        }


        // 3. Fonction pour ouvrir une playlist spécifique

        function openPlaylist(nomPlaylist) {

            playlistsScreen.style.display = 'none';

            tracksScreen.style.display = 'block';

            currentPlaylistTitle.innerText = nomPlaylist;

            

            tracksList.innerHTML = ''; 

            

            const musiques = mesPlaylists[nomPlaylist];

            

            musiques.forEach((musique) => {

                const div = document.createElement('div');

                div.className = 'track';

                div.innerHTML = `<strong>${musique.title}</strong> - ${musique.artist}`;

                div.onclick = () => playTrack(musique.src);

                tracksList.appendChild(div);

            });

        }


        // 4. Fonction pour jouer une musique

        function playTrack(lienAudio) {

            // On encode l'URL pour gérer les espaces et caractères spéciaux dans les noms de fichiers

            audio.src = encodeURI(lienAudio);

            audio.play().catch(e => console.log("Lecture bloquée par le navigateur", e));

        }
